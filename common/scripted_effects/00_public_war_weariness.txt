public_war_weariness_penalty_effect = {
	set_temp_variable = { add_t = penalty_factor }

	set_variable = { war_support_factor_var = add_t }
	multiply_variable = { war_support_factor_var = -0.005 }

	# Force the Update Dynamic Modifier with new Calculated Effects
	force_update_dynamic_modifier = yes
}

conditional_peace_war_calculate_start_effect = {
	ROOT = {
		set_variable = { times_used_propaganda = 0 }
		set_variable = { initial_war_casualties = casualties_k }
		# safeguard
		if = {
			limit = { check_variable = { initial_war_casualties < 1 } }
			set_variable = { initial_war_casualties = 1 }
		}

		set_variable = { war_exhaustion = has_war_support }
		multiply_variable = { war_exhaustion = 100 }

		if = {
			limit = { check_variable = { war_exhaustion < 10 } }
			set_variable = { war_exhaustion = 10 }
		}
	}

	FROM = {
		set_variable = { times_used_propaganda = 0 }
		set_variable = { initial_war_casualties = casualties_k }
		# safeguard
		if = {
			limit = { check_variable = { initial_war_casualties < 1 } }
			set_variable = { initial_war_casualties = 1 }
		}

		set_variable = { war_exhaustion = has_war_support }
		multiply_variable = { war_exhaustion = 100 }
		if = {
			limit = { check_variable = { war_exhaustion < 10 } }
			set_variable = { war_exhaustion = 10 }
		}
	}
}

compare_casualties = {
	set_variable = { THIS.war_casualties = casualties_k }
	subtract_from_variable = { THIS.war_casualties = THIS.initial_war_casualties }

	set_temp_variable = { buf = THIS.war_casualties }
	for_each_scope_loop = {
		array = allies
		add_to_temp_variable = { buf = THIS.war_casualties }
	}

	set_temp_variable = { buf2 = 0 }
	for_each_scope_loop = {
		array = enemies
		add_to_temp_variable = { buf2 = THIS.war_casualties }
	}

	if = {
		limit = {
			check_variable = { buf < 1 }
		}

		set_temp_variable = { buf = 1 }
	}
	if = {
		limit = {
			check_variable = { buf2 < 1 }
		}

		set_temp_variable = { buf2 = 1 }
	}

	log = "[GetDateText]: [ROOT.GetName]: compare_casualties: [?buf]-[?buf2]"

	divide_temp_variable = { buf = buf2 }

	if = {
		limit = { check_variable = { buf > 5 } }
		set_temp_variable = { casulaties_mod = 2 }
	}
	else_if = {
		limit = { check_variable = { buf > 4 } }
		set_temp_variable = { casulaties_mod = 1.75 }
	}
	else_if = {
		limit = { check_variable = { buf > 3 } }
		set_temp_variable = { casulaties_mod = 1.5 }
	}
	else_if = {
		limit = { check_variable = { buf > 2 } }
		set_temp_variable = { casulaties_mod = 1.25 }
	}
	else_if = {
		limit = { check_variable = { buf > 1 } }
		set_temp_variable = { casulaties_mod = 1.1 }
	}
	else_if = {
		limit = { check_variable = { buf > 0.8 } }
		set_temp_variable = { casulaties_mod = 1 }
	}
	else_if = {
		limit = { check_variable = { buf > 0.6 } }
		set_temp_variable = { casulaties_mod = 0.85 }
	}
	else_if = {
		limit = { check_variable = { buf > 0.4 } }
		set_temp_variable = { casulaties_mod = 0.7 }
	}
	else_if = {
		limit = { check_variable = { buf > 0.2 } }
		set_temp_variable = { casulaties_mod = 0.55 }
	}
	else = {
		set_temp_variable = { casulaties_mod = 0.5 }
	}
}

calculate_cap_mod = {
	if = {
		limit = { check_variable = { surender_progress > 0.8 } }
		set_temp_variable = { capitulation_mod = 2 }

	}
	else_if = {
		limit = { check_variable = { surender_progress > 0.6 } }
		set_temp_variable = { capitulation_mod = 1.75 }
	}
	else_if = {
		limit = { check_variable = { surender_progress > 0.4 } }
		set_temp_variable = { capitulation_mod = 1.5 }
	}
	else_if = {
		limit = { check_variable = { surender_progress > 0.2 } }
		set_temp_variable = { capitulation_mod = 1.25 }
	}
}

calculate_monthly_exhaustion = {
	set_temp_variable = { temp_exhaustion = 1 }
	if = {
		limit = { has_offensive_war = yes }
		add_to_temp_variable = { temp_exhaustion = 0.25 }
	}

	set_temp_variable = { casulaties_mod = 1 }
	compare_casualties = yes
	log = "[GetDateText]: [ROOT.GetName]: casulaties_mod: [?casulaties_mod]"

	multiply_temp_variable = { temp_exhaustion = casulaties_mod }

	set_temp_variable = { capitulation_mod = 1 }
	calculate_cap_mod = yes

	multiply_temp_variable = { temp_exhaustion = capitulation_mod }
	log = "[GetDateText]: [ROOT.GetName]: capitulation_mod: [?capitulation_mod]"

	subtract_from_variable = { THIS.war_exhaustion = temp_exhaustion }

	apply_weariness_debuffs = yes
}

apply_weariness_debuffs = {
	if = {
		limit = {
			check_variable = { THIS.war_exhaustion < -50 }
		}

		if = {
			limit = {
				has_idea = war_weariness_penalty_2
			}

			swap_ideas = {
				remove_idea = war_weariness_penalty_2
				add_idea = war_weariness_penalty_3
			}
		}
		else_if = {
			limit = {
				NOT = { has_idea = war_weariness_penalty_3 }
			}

			add_ideas = war_weariness_penalty_3
		}
	}
	else_if = {
		limit = {
			check_variable = { THIS.war_exhaustion < -25 }
		}

		# we increased our support
		if = {
			limit = {
				has_idea = war_weariness_penalty_3
			}

			swap_ideas = {
				remove_idea = war_weariness_penalty_3
				add_idea = war_weariness_penalty_2
			}
		}
		else_if = {
			limit = {
				has_idea = war_weariness_penalty_1
			}

			swap_ideas = {
				remove_idea = war_weariness_penalty_1
				add_idea = war_weariness_penalty_2
			}
		}
		else_if = {
			limit = {
				NOT = { has_idea = war_weariness_penalty_2 }
			}

			add_ideas = war_weariness_penalty_2
		}
	}
	else_if = {
		limit = {
			check_variable = { THIS.war_exhaustion < -0 }
		}

		if = {
			limit = {
				has_idea = war_weariness_penalty_2
			}

			swap_ideas = {
				remove_idea = war_weariness_penalty_2
				add_idea = war_weariness_penalty_1
			}
		}
		else_if = {
			limit = {
				NOT = { has_idea = war_weariness_penalty_1 }
			}

			add_ideas = war_weariness_penalty_1
		}
	}
}
