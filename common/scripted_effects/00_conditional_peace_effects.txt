conditional_peace_war_calculate_start_effect = {
	ROOT = {
		set_variable = { war_casualties = casualties_k }
		# safeguard
		if = {
			limit = { check_variable = { war_casualties < 1 } }
			set_variable = { war_casualties = 1 }
		}
	}
	set_temp_variable = { res = 0 }
	calculate_war_cost_for_attacker = yes
	set_temp_variable = { attacker_cost = yes }
	calculate_war_support_for_defender = yes

	FROM = {
		set_variable = { war_casualties = casualties_k }
		# safeguard
		if = {
			limit = { check_variable = { war_casualties < 1 } }
			set_variable = { war_casualties = 1 }
		}
	}
	set_temp_variable = { res = 0 }
	calculate_war_cost_for_defender = yes
	set_temp_variable = { defender_cost = yes }
	calculate_war_support_for_attacker = yes
}

calculate_war_cost_for_attacker = {
	set_temp_variable = { buf = ROOT.num_of_claimed_cores }
	divide_temp_variable = { buf = num_controlled_states }
	multiply_temp_variable = { buf = 20 }
	add_to_temp_variable = { res = buf }

	set_temp_variable = { buf = ROOT.num_of_claimed_states }
	divide_temp_variable = { buf = num_controlled_states }
	multiply_temp_variable = { buf = 100 }
	add_to_temp_variable = { res = buf }

	if = {
		limit = {
			is_in_array = {
				array = ROOT.war_targets
				value = 3
			}
		}

		add_to_temp_variable = { res = 35 }
	}

	if = {
		limit = {
			is_in_array = {
				array = ROOT.war_targets
				value = 4
			}
		}

		add_to_temp_variable = { res = 60 }
	}

	if = {
		limit = {
			is_in_array = {
				array = ROOT.war_targets
				value = 5
			}
		}

		add_to_temp_variable = { res = 10 }
	}

	if = {
		limit = {
			is_in_array = {
				array = ROOT.war_targets
				value = 6
			}
		}

		add_to_temp_variable = { res = 50 }
	}

	if = {
		limit = {
			is_in_array = {
				array = ROOT.war_targets
				value = 7
			}
		}

		add_to_temp_variable = { res = 30 }
	}
}

calculate_war_support_for_defender = {
	set_temp_variable = { temp_exhaustion = has_war_support }
	multiply_temp_variable = { temp_exhaustion = res }
	FROM = {
		set_variable = { war_exhaustion = temp_exhaustion }
	}
}

calculate_war_support_for_attacker = {
	set_temp_variable = { temp_exhaustion = has_war_support }
	multiply_temp_variable = { temp_exhaustion = res }
	ROOT = {
		set_variable = { war_exhaustion = temp_exhaustion }
	}
}

compare_casualties = {
	set_temp_variable = { buf = THIS.war_casualties }
	divide_temp_variable = { buf = opposing_tag.war_casualties }
	if = {
		limit = { buf > 5 }
		set_temp_variable = { casulaties_mod = 2 }
	}
	else_if = {
		limit = { buf > 4 }
		set_temp_variable = { casulaties_mod = 1.75 }
	}
	else_if = {
		limit = { buf > 3 }
		set_temp_variable = { casulaties_mod = 1.5 }
	}
	else_if = {
		limit = { buf > 2 }
		set_temp_variable = { casulaties_mod = 1.25 }
	}
	else_if = {
		limit = { buf > 1 }
		set_temp_variable = { casulaties_mod = 1.1 }
	}
	else_if = {
		limit = { buf > 0.8 }
		set_temp_variable = { casulaties_mod = 0.9 }
	}
	else_if = {
		limit = { buf > 0.6 }
		set_temp_variable = { casulaties_mod = 0.8 }
	}
	else_if = {
		limit = { buf > 0.4 }
		set_temp_variable = { casulaties_mod = 0.7 }
	}
	else_if = {
		limit = { buf > 0.2 }
		set_temp_variable = { casulaties_mod = 0.6 }
	}
	else = {
		set_temp_variable = { casulaties_mod = 0.5 }
	}
}

calculate_cap_mod = {
	if = {
		limit = { surrender_progress > 0.8 }
		set_temp_variable = { capitulation_mod = 2 }
	}
	else_if = {
		limit = { surrender_progress > 0.6 }
		set_temp_variable = { capitulation_mod = 1.75 }
	}
	else_if = {
		limit = { surrender_progress > 0.4 }
		set_temp_variable = { capitulation_mod = 1.5 }
	}
	else_if = {
		limit = { surrender_progress > 0.2 }
		set_temp_variable = { capitulation_mod = 1.25 }
	}
}

calculate_monthly_exhaustion = {
	set_temp_variable = { temp_exhaustion = 1 }
	if = {
		limit = { has_offensive_war = yes }
		add_to_temp_variable = { temp_exhaustion = 0.5 }
	}

	set_temp_variable = { opposing_tag = THIS.war_target_tag }
	set_temp_variable = { casulaties_mod = 1 }
	compare_casualties = yes

	multiply_temp_variable = { temp_exhaustion = casulaties_mod }

	set_temp_variable = { capitulation_mod = 1 }
	calculate_cap_mod = yes

	multiply_temp_variable = { temp_exhaustion = capitulation_mod }

	subtract_from_variable = { THIS.war_exhaustion = temp_exhaustion }
}
